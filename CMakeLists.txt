cmake_minimum_required(VERSION 3.20)
project(seastar_python_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Seastar REQUIRED)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
add_library(probe_runtime STATIC native/runtime.cc native/work.cc native/server.cc native/http1.cc)
target_link_libraries(probe_runtime PUBLIC Seastar::seastar ZLIB::ZLIB Threads::Threads)
target_compile_definitions(probe_runtime PUBLIC SEASTAR_DEFAULT_ALLOCATOR)
# The package's native extension, oldman_server.native._native, built in place.
pybind11_add_module(_native native/bridge.cc)
target_link_libraries(_native PRIVATE probe_runtime)
set_target_properties(_native PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/src/oldman_server/native)
add_executable(seastar_baseline native/baseline.cc)
target_link_libraries(seastar_baseline PRIVATE probe_runtime)
add_executable(http_load native/http_load.cc)
target_link_libraries(http_load PRIVATE probe_runtime)
add_executable(fixed_native probes/fixed/native_main.cc)
target_link_libraries(fixed_native PRIVATE probe_runtime)
target_include_directories(fixed_native PRIVATE native)
add_executable(async_native probes/async_business/native_main.cc)
target_link_libraries(async_native PRIVATE probe_runtime)
target_include_directories(async_native PRIVATE native)
add_executable(io_native probes/io_business/native_main.cc)
target_link_libraries(io_native PRIVATE probe_runtime)
target_include_directories(io_native PRIVATE native)
add_executable(framework_native probes/framework_json/native_main.cc)
target_link_libraries(framework_native PRIVATE probe_runtime)
target_include_directories(framework_native PRIVATE native)
add_executable(sanic_baseline_native probes/sanic_comparison/native.cc)
target_link_libraries(sanic_baseline_native PRIVATE Seastar::seastar Threads::Threads)
target_compile_definitions(sanic_baseline_native PRIVATE SEASTAR_DEFAULT_ALLOCATOR)
pybind11_add_module(_binding_probe probes/bindings/probe.cc)
target_link_libraries(_binding_probe PRIVATE probe_runtime)
target_include_directories(_binding_probe PRIVATE native)
